cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(mouse_radio_logger)
target_sources(app PRIVATE src/main.c)
option(DISCOVERY_PROBE "Bounded experimental discovery and data ACK probe" OFF)
option(WIRELESS_HID "Bounded wireless mouse USB HID output" OFF)
option(NORMAL_DONGLE "Automatic continuous receiver without a serial reader" OFF)
option(LATENCY_DIAGNOSTICS "Quiet fixed-memory radio/USB timing with stopped-only dump" OFF)
option(FAST_POLL_DIAGNOSTICS "Compare 10us polling in the timing image only" OFF)
option(INPUT_CONSERVATION "Compare accepted mouse input totals with host capture" OFF)
option(USB_PHASE_DIAGNOSTICS "Hardware SOF phase with accepted input accounting" OFF)
option(RF_CONFIG_READ_DIAGNOSTICS "One fixed RF version read without pointer output" OFF)
option(RF_CONFIG_HID_DIAGNOSTICS "Host-triggered Feature report version read with prototype identity" OFF)
option(AUX_HID_FORWARDING "Auxiliary USB HID forwarding for keyboard, consumer, system and vendor notifications" OFF)
if(AUX_HID_FORWARDING AND NOT NORMAL_DONGLE)
  message(FATAL_ERROR "Auxiliary HID forwarding requires the normal dongle startup path")
endif()
if(RF_CONFIG_HID_DIAGNOSTICS AND NOT RF_CONFIG_READ_DIAGNOSTICS)
  message(FATAL_ERROR "Config HID requires the read-only RF diagnostic")
endif()
if(RF_CONFIG_READ_DIAGNOSTICS AND (NOT LATENCY_DIAGNOSTICS OR USB_PHASE_DIAGNOSTICS))
  message(FATAL_ERROR "Config read requires quiet timing transport, without phase diagnostics")
endif()
if(USB_PHASE_DIAGNOSTICS AND (NOT LATENCY_DIAGNOSTICS OR NOT INPUT_CONSERVATION))
  message(FATAL_ERROR "USB phase requires latency and input diagnostics")
endif()
if(INPUT_CONSERVATION AND NOT LATENCY_DIAGNOSTICS)
  message(FATAL_ERROR "Input conservation requires latency diagnostics")
endif()
if(FAST_POLL_DIAGNOSTICS AND NOT LATENCY_DIAGNOSTICS)
  message(FATAL_ERROR "Fast polling is only an instrumented diagnostic experiment")
endif()
if(LATENCY_DIAGNOSTICS AND NOT NORMAL_DONGLE)
  message(FATAL_ERROR "Latency diagnostics requires the normal dongle startup path")
endif()
if(NORMAL_DONGLE AND NOT WIRELESS_HID)
  message(FATAL_ERROR "Normal dongle requires wireless HID")
endif()
if(WIRELESS_HID AND NOT DISCOVERY_PROBE)
  message(FATAL_ERROR "Wireless HID requires the bounded discovery probe")
endif()
if(DISCOVERY_PROBE)
  target_compile_definitions(app PRIVATE DISCOVERY_PROBE=1)
endif()
if(WIRELESS_HID)
  target_compile_definitions(app PRIVATE WIRELESS_HID=1)
endif()
if(NORMAL_DONGLE)
  target_compile_definitions(app PRIVATE NORMAL_DONGLE=1)
endif()
if(LATENCY_DIAGNOSTICS)
  target_compile_definitions(app PRIVATE LATENCY_DIAGNOSTICS=1)
endif()
if(FAST_POLL_DIAGNOSTICS)
  target_compile_definitions(app PRIVATE FAST_POLL_DIAGNOSTICS=1)
endif()
if(INPUT_CONSERVATION)
  target_compile_definitions(app PRIVATE INPUT_CONSERVATION=1)
endif()

if(USB_PHASE_DIAGNOSTICS)
  target_compile_definitions(app PRIVATE USB_PHASE_DIAGNOSTICS=1)
endif()
if(RF_CONFIG_READ_DIAGNOSTICS)
  target_compile_definitions(app PRIVATE RF_CONFIG_READ_DIAGNOSTICS=1)
endif()
if(RF_CONFIG_HID_DIAGNOSTICS)
  target_compile_definitions(app PRIVATE RF_CONFIG_HID_DIAGNOSTICS=1)
endif()
if(AUX_HID_FORWARDING)
  target_compile_definitions(app PRIVATE AUX_HID_FORWARDING=1)
endif()
